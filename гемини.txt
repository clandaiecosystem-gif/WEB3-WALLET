abstract contract ReentrancyGuard {
    uint256 private _locked = 1;
    modifier nonReentrant() {
        require(_locked == 1, "reentrant");
        _locked = 2;
        _;
        _locked = 1;
    }
}

/* =========================================================================
   SMART WALLET GUARD v5.1.8
   =========================================================================
   Правки относительно v5.1.7:

   MUST-FIX:
     1. acceptMigration: cross-check теперь БЕЗУСЛОВНЫЙ для subscription,
        treasury, appVersionHash, P2P config. Требуется точное совпадение
        с raw-состоянием old guard. Предотвращает потерю state через
        подписанную, но некорректную migrationData (например, subscription
        со сброшенным expiresAt=0 при активной подписке).
     2. Добавлено state `address public migratedTo`. Устанавливается в
        finalizeMigration. `proposeWalletGuardMigration` больше не имеет
        модификатора `notMigrated` — можно подавать wallet proposals
        ПОСЛЕ finalize, используя `migratedTo` как утверждённый newGuard.
        Устраняет требование успеть со всеми wallets в 48h окне.
     3. `_checkWalletGuard(w)` добавлен в `emergencySweep` для
        консистентности с transfer/transferNative/executeSwap/subscribe.

   DOCUMENTATION:
     - executeSwap whitelist для (recipient, tokenOut) — сознательное
       расширение ТЗ §23 для жёсткого лимита на получателя swap.
       Клиент обязан ставить minAmountOut с учётом возможного overshoot:
       если remaining = R, а favorable slippage даёт actualOut > R,
       транзакция ревертится (safe-by-default). Это гарантирует, что
       whitelist limit никогда не будет превышен.
     - После wallet migration old guard продолжает хранить wallet в
       реестре (исторический snapshot). Все операции old guard
       блокируются через `notMigrated`. Удаление wallet из old реестра
       невозможно (immutable contract).
     - subscribe требует, чтобы wallet уже был мигрирован на текущий
       guard. Последовательность: wallet migration → enablePermit2 →
       subscribe.

   Открытые пункты ТЗ (требуют обновления):
     §63 — emergency migration сохраняет 48h delay
     §64 — adapter policies не мигрируются (immutable guard)
     §82 — events семантически эквивалентны, имена иные
     §100 — permission bits зарезервированы для будущих модулей
     §113 — getWhitelist() невозможен on-chain (mapping)
     §114 — VERSION = "5.1.8"
     §7 — nonce scope для emergencySweep = wallet, для subscribe = fromWallet
   ========================================================================= */

        address _expectedOldGuard

        require(_mainWallet != address(0), "zero main");
        require(_owners.length == TOTAL_OWNERS, "need 5");
        require(_isHot.length == TOTAL_OWNERS, "need 5 flags");
        require(address(_permit2) != address(0), "zero permit2");

        mainWallet = _mainWallet;
        permit2 = address(_permit2);
        expectedOldGuard = _expectedOldGuard;

        DOMAIN_SEPARATOR = keccak256(abi.encode(
            DOMAIN_TYPEHASH,
            keccak256("SmartWalletGuard"),
            keccak256(bytes(VERSION)),
            block.chainid,
            address(this)
        ));

        uint256 hotCount;
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            address o = _owners[i];
            require(o != address(0), "zero owner");
            require(!ownerInfo[o].isOwner, "dup");
            ownerInfo[o] = OwnerInfo(true, _isHot[i]);
            owners.push(o);
            if (_isHot[i]) hotCount++;
        }
        require(hotCount == HOT_OWNERS, "need 3 hot");
    }

    // =========================================================================
    // EIP-712 helpers
    // =========================================================================

    function _hashTypedData(bytes32 structHash) internal view returns (bytes32) {
        return keccak256(abi.encodePacked("\x19\x01", DOMAIN_SEPARATOR, structHash));
    }
    function _hashAddressArray(address[] calldata values) internal pure returns (bytes32) {
        bytes memory e = new bytes(values.length * 32);
        for (uint256 i = 0; i < values.length; i++) {
            uint256 v = uint256(uint160(values[i]));
            assembly { mstore(add(add(e, 32), mul(i, 32)), v) }
        }
        return keccak256(e);
    }
    function _hashBoolArray(bool[] calldata values) internal pure returns (bytes32) {
        bytes memory e = new bytes(values.length * 32);
        for (uint256 i = 0; i < values.length; i++) {
            uint256 v = values[i] ? 1 : 0;
            assembly { mstore(add(add(e, 32), mul(i, 32)), v) }
        }
        return keccak256(e);
    }
    function _recover(bytes32 digest, bytes calldata sig) internal pure returns (address) {
        require(sig.length == 65, "bad sig length");
        bytes32 r; bytes32 s; uint8 v;
        assembly {
            r := calldataload(sig.offset)
            s := calldataload(add(sig.offset, 32))
            v := byte(0, calldataload(add(sig.offset, 64)))
        }
        if (v < 27) v += 27;
        require(v == 27 || v == 28, "bad v");
        require(uint256(s) <= uint256(SECP256K1N_HALF), "high-s");
        address signer = ecrecover(digest, v, r, s);
        require(signer != address(0), "invalid sig");
        return signer;
    }
    function _verifySignatures(
        bytes32 digest, bytes[] calldata sigs, uint256 threshold, bool requireCold
    ) internal view returns (bool) {
        if (sigs.length < threshold) return false;
        require(sigs.length <= TOTAL_OWNERS, "too many sigs");
        address[] memory seen = new address[](sigs.length);
        uint256 coldValid;
        for (uint256 i = 0; i < sigs.length; i++) {
            address s = _recover(digest, sigs[i]);
            OwnerInfo memory info = ownerInfo[s];
            if (!info.isOwner) return false;
            for (uint256 j = 0; j < i; j++) {
                if (seen[j] == s) return false;
            }
            seen[i] = s;
            if (!info.isHot) coldValid++;
        }
        if (requireCold && coldValid == 0) return false;
        return true;
    }

    function _consumeNonce(address w, uint256 n) internal {
        require(!usedNonces[w][n], "nonce used");
        usedNonces[w][n] = true;
    }
    function _checkDeadline(uint256 d) internal view { require(block.timestamp <= d, "expired"); }
    function _checkPermit2Allowance(address w, address token, uint256 amount) internal view {
        (uint160 allowed, uint48 exp, ) = IPermit2(permit2).allowance(w, token, address(this));
        require(uint256(allowed) >= amount, "insufficient permit2 allowance");
        require(uint256(exp) >= block.timestamp, "permit2 allowance expired");
    }
    function _checkNotPaused(address w) internal view {
        require(!ISmartWallet(w).paused(), "wallet paused");
    }
    function _checkRecipient(address r) internal view {
        require(r != address(0), "zero recipient");
        require(!blacklist[r], "blacklisted");
    }
    function _isRegisteredWallet(address w) internal view returns (bool) {
        return hotWallets[w] || coldWallets[w];
    }
    function _checkWalletGuard(address w) internal view {
        require(ISmartWallet(w).guard() == address(this), "wallet not ours");
    }

    function _planMask(SubscriptionPlan p) internal pure returns (uint256) {
        if (p == SubscriptionPlan.FREE) {
            return PERM_APP_VERIFY | PERM_EMERGENCY;
        }
        return PERM_APP_VERIFY | PERM_EMERGENCY | PERM_COLD_WARM
             | PERM_P2P | PERM_SECURITY | PERM_TRADING;
    }

    function activePlan() public view returns (SubscriptionPlan) {
        Subscription memory s = subscriptions[mainWallet];
        if (s.expiresAt == 0 || block.timestamp >= s.expiresAt) return SubscriptionPlan.FREE;
        return s.plan;
    }

    function hasPermission(uint256 perm) public view returns (bool) {
        return (_planMask(activePlan()) & perm) != 0;
    }
    function hasPermission(address account, uint256 perm) external view returns (bool) {
        if (account != mainWallet) return false;
        return (_planMask(activePlan()) & perm) != 0;
    }

    function getSubscription() external view returns (SubscriptionPlan plan, uint64 expiresAt) {
        Subscription memory s = subscriptions[mainWallet];
        expiresAt = s.expiresAt;
        plan = activePlan();
    }
    function getRawSubscription() external view returns (uint8 plan, uint64 expiresAt) {
        Subscription memory s = subscriptions[mainWallet];
        return (uint8(s.plan), s.expiresAt);
    }
    function getPlanConfig(SubscriptionPlan plan)
        external view
        returns (address paymentToken, uint256 price, uint64 period, bool enabled)
    {
        PlanConfig memory c = planConfigs[plan];
        return (c.paymentToken, c.price, c.period, c.enabled);
    }

    function transfer(
        address hotWallet, address token, address recipient, uint256 amount,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(hotWallets[hotWallet], "not hot wallet");
        require(amount > 0 && amount <= type(uint160).max, "bad amount");
        _checkWalletGuard(hotWallet);
        _checkRecipient(recipient);
        require(!blacklist[token], "token blacklisted");
        require(approvedToken[token], "token not approved");
        _checkNotPaused(hotWallet);
        _checkDeadline(deadline);
        _consumeNonce(hotWallet, nonce);

        WhitelistEntry storage w = whitelist[recipient][token];
        require(w.exists, "not whitelisted");
        require(block.timestamp <= w.deadline, "whitelist deadline expired");
        require(w.remaining >= amount, "limit exceeded");

        _checkPermit2Allowance(hotWallet, token, amount);

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            TRANSFER_TYPEHASH, hotWallet, token, recipient, amount, nonce, deadline)));
        require(_verifySignatures(digest, signatures, HOT_THRESHOLD, false), "invalid sigs");

        w.remaining -= amount;
        IPermit2(permit2).transferFrom(hotWallet, recipient, uint160(amount), token);
        emit TransferExecuted(hotWallet, token, recipient, amount, nonce);
    }

    function transferNative(
        address hotWallet, address payable recipient, uint256 amount,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(hotWallets[hotWallet], "not hot wallet");
        require(amount > 0, "bad amount");
        _checkWalletGuard(hotWallet);
        _checkRecipient(recipient);
        _checkNotPaused(hotWallet);
        _checkDeadline(deadline);
        _consumeNonce(hotWallet, nonce);

        WhitelistEntry storage w = whitelist[recipient][NATIVE];
        require(w.exists, "native not whitelisted");
        require(block.timestamp <= w.deadline, "whitelist deadline expired");
        require(w.remaining >= amount, "limit exceeded");

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            NATIVE_TRANSFER_TYPEHASH, hotWallet, recipient, amount, nonce, deadline)));
        require(_verifySignatures(digest, signatures, HOT_THRESHOLD, false), "invalid sigs");

        w.remaining -= amount;
        ISmartWallet(hotWallet).sendNativeByGuard(recipient, amount);
        emit NativeTransferExecuted(hotWallet, recipient, amount, nonce);
    }

    // =========================================================================
    // Swap (2/3 Hot)
    // =========================================================================

    /// @dev Whitelist для (recipient, tokenOut) — сознательное расширение ТЗ §23
    ///      для жёсткого лимита на получателя swap. Пост-условие
    ///      `actualOut <= remaining` гарантирует, что лимит не превышается
    ///      даже при favorable slippage. Клиент обязан ставить minAmountOut
    ///      с учётом возможного overshoot.
    function executeSwap(
        address hotWallet, address adapter, address tokenIn, address tokenOut,
        uint256 amountIn, uint256 minAmountOut, address recipient, bytes calldata adapterData,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(hotWallets[hotWallet], "not hot wallet");
        require(amountIn > 0 && amountIn <= type(uint160).max, "bad amount");
        require(tokenIn != address(0) && tokenOut != address(0), "zero token");
        require(tokenIn != tokenOut, "same token");
        require(adapter.code.length > 0, "adapter not contract");
        _checkWalletGuard(hotWallet);
        _checkRecipient(recipient);
        require(!blacklist[adapter], "adapter blacklisted");
        require(!blacklist[tokenIn], "tokenIn blacklisted");
        require(!blacklist[tokenOut], "tokenOut blacklisted");
        require(approvedToken[tokenIn], "tokenIn not approved");
        require(approvedToken[tokenOut], "tokenOut not approved");

        WhitelistEntry storage wr = whitelist[recipient][tokenOut];
        require(wr.exists, "recipient not whitelisted for tokenOut");
        require(block.timestamp <= wr.deadline, "recipient deadline expired");
        require(wr.remaining >= minAmountOut, "recipient limit exceeded");

        require(ISwapAdapter(adapter).guard() == address(this), "wrong adapter guard");
        address router = ISwapAdapter(adapter).router();
        require(approvedRouter[router], "router not approved");
        require(!blacklist[router], "router blacklisted");

        _checkNotPaused(hotWallet);
        _checkDeadline(deadline);
        _consumeNonce(hotWallet, nonce);

        AdapterPolicy storage p = adapterPolicy[adapter][tokenIn];
        require(p.exists, "adapter not whitelisted");
        require(block.timestamp <= p.deadline, "adapter deadline expired");
        require(p.remaining >= amountIn, "adapter limit exceeded");

        _checkPermit2Allowance(hotWallet, tokenIn, amountIn);

        uint256 beforeOut = IERC20(tokenOut).balanceOf(recipient);
        bytes32 adapterDataHash = keccak256(adapterData);
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SWAP_TYPEHASH, hotWallet, adapter, tokenIn, tokenOut, amountIn,
            minAmountOut, recipient, adapterDataHash, nonce, deadline)));
        require(_verifySignatures(digest, signatures, HOT_THRESHOLD, false), "invalid sigs");

        p.remaining -= amountIn;
        IPermit2(permit2).transferFrom(hotWallet, adapter, uint160(amountIn), tokenIn);
        ISwapAdapter(adapter).swap(tokenIn, tokenOut, amountIn, minAmountOut, recipient, deadline, adapterData);

        uint256 actualOut = IERC20(tokenOut).balanceOf(recipient) - beforeOut;
        require(actualOut >= minAmountOut, "insufficient output");
        require(actualOut <= wr.remaining, "recipient limit exceeded (post)");
        wr.remaining -= actualOut;

        emit SwapExecuted(hotWallet, adapter, tokenIn, tokenOut, amountIn, actualOut, recipient, nonce);
    }

    /// @dev v5.1.8: `_checkWalletGuard` добавлен для консистентности.
    function emergencySweep(
        address w, address token, uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(_isRegisteredWallet(w), "unknown wallet");
        _checkWalletGuard(w); // v5.1.8
        require(ISmartWallet(w).paused(), "not paused");
        if (token != NATIVE) require(!blacklist[token], "token blacklisted");

        RecoveryTarget memory rt = recoveryTargets[w][token];
        require(rt.exists, "no recovery target");
        require(!blacklist[rt.recipient], "recovery recipient blacklisted");

        _checkDeadline(deadline);
        _consumeNonce(w, nonce);

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            EMERGENCY_SWEEP_TYPEHASH, w, token, rt.recipient, nonce, deadline)));
        require(_verifySignatures(digest, signatures, HOT_THRESHOLD, false), "invalid emergency sweep sigs");

        uint256 amount;
        if (token == NATIVE) {
            amount = w.balance;
            require(amount > 0, "zero native");
            ISmartWallet(w).sendNativeByGuard(payable(rt.recipient), amount);
        } else {
            amount = IERC20(token).balanceOf(w);
            require(amount > 0, "zero token");
            ISmartWallet(w).emergencyEnablePermit2ForGuard(token);
            _checkPermit2Allowance(w, token, amount);
            IPermit2(permit2).transferFrom(w, rt.recipient, uint160(amount), token);
        }
        emit EmergencySweepExecuted(w, token, rt.recipient, amount);
    }

    // =========================================================================
    // Change owners (3/5 + cold)
    // =========================================================================

    function changeOwners(
        address[] calldata newOwners, bool[] calldata newIsHot,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newOwners.length == TOTAL_OWNERS, "need 5");
        require(newIsHot.length == TOTAL_OWNERS, "need 5 flags");

        uint256 hotCount;
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            require(newOwners[i] != address(0), "zero owner");
            for (uint256 j = 0; j < i; j++) require(newOwners[j] != newOwners[i], "dup");
            if (newIsHot[i]) hotCount++;
        }
        require(hotCount == HOT_OWNERS, "need 3 hot");

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            OWNERS_TYPEHASH, mainWallet,
            _hashAddressArray(newOwners), _hashBoolArray(newIsHot),
            nonce, deadline)));
        require(_verifySignatures(digest, signatures, CRITICAL_THRESHOLD, true), "invalid critical sigs");

        for (uint256 i = 0; i < owners.length; i++) {
            ownerInfo[owners[i]] = OwnerInfo(false, false);
            emit OwnerChanged(owners[i], false, false);
        }
        delete owners;
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            ownerInfo[newOwners[i]] = OwnerInfo(true, newIsHot[i]);
            owners.push(newOwners[i]);
            emit OwnerChanged(newOwners[i], newIsHot[i], true);
        }
    }

    function proposeMigration(
        address newGuard, bytes32 migrationDataHash,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newGuard != address(0), "zero");
        require(newGuard != address(this), "same");
        require(newGuard.code.length > 0, "not contract");
        require(!pendingMigration.pending, "pending");
        require(
            migrationCancelCount[mainWallet][newGuard] < MAX_MIGRATION_CANCELS,
            "max cancels reached, use emergency path"
        );

        uint256 executeAfter = block.timestamp + MIGRATION_DELAY;
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            MIGRATION_TYPEHASH, mainWallet, newGuard, migrationDataHash, nonce, executeAfter, deadline)));
        require(_verifySignatures(digest, signatures, CRITICAL_THRESHOLD, true), "invalid migration sigs");

        pendingMigration = Migration(newGuard, migrationDataHash, block.timestamp, executeAfter, 0, true);
        emit MigrationProposed(address(this), newGuard, migrationDataHash, executeAfter);
    }

    function cancelMigration(uint256 nonce, uint256 deadline, bytes[] calldata signatures) external notMigrated {
        require(pendingMigration.pending, "no migration");
        require(block.timestamp < pendingMigration.executeAfter, "timelock passed");
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        address newGuard = pendingMigration.newGuard;

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            CANCEL_MIGRATION_TYPEHASH, mainWallet, newGuard, nonce, deadline)));
        require(_verifySignatures(digest, signatures, CRITICAL_THRESHOLD, true), "invalid cancel sigs");

        uint256 newCount = migrationCancelCount[mainWallet][newGuard] + 1;
        migrationCancelCount[mainWallet][newGuard] = newCount;

        delete pendingMigration;
        emit MigrationCancelled(address(this), newGuard, newCount);
    }

    function proposeMigrationWithEmergency(
        address newGuard, bytes32 migrationDataHash,
        uint256 nonce, uint256 deadline,
        bytes calldata emergencySignature, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newGuard != address(0), "zero");
        require(newGuard != address(this), "same");
        require(newGuard.code.length > 0, "not contract");
        require(!pendingMigration.pending, "pending");
        require(migrationCancelCount[mainWallet][newGuard] >= MAX_MIGRATION_CANCELS, "not unlocked");

        uint256 executeAfter = block.timestamp + MIGRATION_DELAY;
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            MIGRATION_EMERGENCY_TYPEHASH, mainWallet, newGuard, migrationDataHash, nonce, executeAfter, deadline)));

        if (migrationEmergencyKey != address(0)) {
            require(
                _recover(digest, emergencySignature) == migrationEmergencyKey,
                "invalid emergency key"
            );
        }
        require(_verifySignatures(digest, signatures, CRITICAL_THRESHOLD, true), "invalid migration sigs");

        pendingMigration = Migration(newGuard, migrationDataHash, block.timestamp, executeAfter, 0, true);
        emit MigrationProposed(address(this), newGuard, migrationDataHash, executeAfter);
    }

    /// @dev v5.1.8: сохраняет migratedTo для последующего proposeWalletGuardMigration.
    function finalizeMigration(bytes calldata migrationData) external nonReentrant notMigrated {
        require(pendingMigration.pending, "no migration");
        require(block.timestamp >= pendingMigration.executeAfter, "timelock active");
        bytes32 dataHash = keccak256(migrationData);
        require(dataHash == pendingMigration.migrationDataHash, "data hash mismatch");

        address newGuard = pendingMigration.newGuard;
        migratedOut = true;
        migratedTo = newGuard; // v5.1.8
        delete pendingMigration;

        IGuardMigration(newGuard).acceptMigration(address(this), mainWallet, dataHash, migrationData);
        emit MigrationFinalized(address(this), newGuard);
        emit MigratedOut(newGuard);
    }

    // ---------- Wallet migration helpers ----------

    /// @dev v5.1.8: работает и до, и после finalizeMigration.
    ///      До finalize: newGuard == pendingMigration.newGuard.
    ///      После finalize: newGuard == migratedTo.
    function proposeWalletGuardMigration(address wallet, address newGuard)
        external nonReentrant
    {
        address approvedNewGuard;
        if (pendingMigration.pending) {
            approvedNewGuard = pendingMigration.newGuard;
        } else if (migratedOut && migratedTo != address(0)) {
            approvedNewGuard = migratedTo;
        } else {
            revert("no approved migration");
        }
        require(newGuard == approvedNewGuard, "guard != approved");
        require(_isRegisteredWallet(wallet), "unknown wallet");
        require(wallet.code.length > 0, "wallet not contract");
        require(ISmartWallet(wallet).guard() == address(this), "not our wallet");
        require(ISmartWallet(wallet).permit2() == permit2, "wrong permit2");

        ISmartWallet(wallet).proposeGuardMigration(newGuard);
        emit WalletGuardMigrationProposed(wallet, newGuard);
    }

    function acceptWalletGuardMigration(address wallet) external nonReentrant {
        require(migrationAccepted, "not migrated in");
        require(!migratedOut, "migrated out");
        require(_isRegisteredWallet(wallet), "unknown wallet");
        require(wallet.code.length > 0, "wallet not contract");

        require(ISmartWallet(wallet).guard() == expectedOldGuard, "wallet not old guard");
        ISmartWallet(wallet).acceptGuardMigration();
        require(ISmartWallet(wallet).guard() == address(this), "wallet migration failed");

        emit WalletGuardMigrationAccepted(wallet);
    }

    function enablePermit2OnWallet(address wallet, address token) external nonReentrant {
        require(migrationAccepted, "not migrated in");
        require(!migratedOut, "migrated out");
        require(_isRegisteredWallet(wallet), "unknown wallet");
        require(ISmartWallet(wallet).guard() == address(this), "wallet not migrated");
        require(approvedToken[token], "token not approved");
        require(token != NATIVE, "native uses no permit2");

        ISmartWallet(wallet).emergencyEnablePermit2ForGuard(token);
        emit WalletPermit2Enabled(wallet, token);

    function subscribe(
        SubscriptionPlan plan, address fromWallet,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(plan == SubscriptionPlan.STANDARD || plan == SubscriptionPlan.PREMIUM, "bad plan");
        require(treasury != address(0), "treasury not set");

        PlanConfig memory cfg = planConfigs[plan];
        require(cfg.enabled, "plan disabled");
        require(cfg.paymentToken != address(0), "no payment token");
        require(cfg.price > 0, "no price");
        require(cfg.period > 0, "no period");
        require(approvedToken[cfg.paymentToken], "payment token not approved");
        require(!blacklist[cfg.paymentToken], "payment token blacklisted");

        require(_isRegisteredWallet(fromWallet), "bad wallet");
        _checkWalletGuard(fromWallet);
        _checkNotPaused(fromWallet);
        _checkDeadline(deadline);
        _consumeNonce(fromWallet, nonce);
        _checkPermit2Allowance(fromWallet, cfg.paymentToken, cfg.price);

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SUBSCRIBE_TYPEHASH, mainWallet, uint8(plan),
            cfg.paymentToken, cfg.price, fromWallet, nonce, deadline)));
        require(_verifySignatures(digest, signatures, HOT_THRESHOLD, false), "invalid sigs");

        Subscription memory s = subscriptions[mainWallet];
        uint64 newExpiry;
        if (s.expiresAt > block.timestamp && s.plan == plan) {
            newExpiry = s.expiresAt + cfg.period;
        } else {
            newExpiry = uint64(block.timestamp) + cfg.period;
        }
        subscriptions[mainWallet] = Subscription(plan, newExpiry);

        IPermit2(permit2).transferFrom(fromWallet, treasury, uint160(cfg.price), cfg.paymentToken);
        emit Subscribed(mainWallet, plan, cfg.paymentToken, cfg.price, newExpiry);
    }

    /// @dev v5.1.8: cross-check БЕЗУСЛОВНЫЙ для subscription, treasury,
    ///      appVersionHash, P2P config. Требуется точное совпадение с
    ///      raw-состоянием old guard. Предотвращает потерю state через
    ///      подписанную, но некорректную migrationData.
    function acceptMigration(
        address oldGuard,
        address expectedMainWallet,
        bytes32 migrationDataHash,
        bytes calldata migrationData
    ) external override nonReentrant {
        require(!migrationAccepted, "already accepted");
        require(!migratedOut, "already migrated out");
        require(expectedOldGuard != address(0), "not a migration target");
        require(oldGuard == expectedOldGuard, "unexpected old guard");
        require(msg.sender == expectedOldGuard, "not old guard");
        require(expectedMainWallet == mainWallet, "wrong main wallet");
        require(keccak256(migrationData) == migrationDataHash, "hash mismatch");
        require(IGuardState(oldGuard).mainWallet() == mainWallet, "old main mismatch");
        require(IGuardState(oldGuard).permit2() == permit2, "permit2 mismatch");
        require(IGuardState(oldGuard).migratedOut(), "old guard not migrated");

        MigrationData memory d = abi.decode(migrationData, (MigrationData));

        // ---- Owners ----
        require(d.newOwners.length == TOTAL_OWNERS && d.newIsHot.length == TOTAL_OWNERS, "bad owners");
        uint256 hotCount;
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            require(d.newOwners[i] != address(0), "zero owner");
            for (uint256 j = 0; j < i; j++) require(d.newOwners[j] != d.newOwners[i], "dup");
            if (d.newIsHot[i]) hotCount++;
        }
        require(hotCount == HOT_OWNERS, "need 3 hot");

        address[] memory oldOwners = IGuardState(oldGuard).getOwners();
        require(oldOwners.length == TOTAL_OWNERS, "old owners mismatch");
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            require(oldOwners[i] == d.newOwners[i], "owner mismatch");
            require(IGuardState(oldGuard).isOwnerHot(oldOwners[i]) == d.newIsHot[i], "isHot mismatch");
        }
        for (uint256 i = 0; i < owners.length; i++) {
            ownerInfo[owners[i]] = OwnerInfo(false, false);
            emit OwnerChanged(owners[i], false, false);
        }
        delete owners;
        for (uint256 i = 0; i < TOTAL_OWNERS; i++) {
            ownerInfo[d.newOwners[i]] = OwnerInfo(true, d.newIsHot[i]);
            owners.push(d.newOwners[i]);
            emit OwnerChanged(d.newOwners[i], d.newIsHot[i], true);
        }

        // ---- Hot wallets ----
        for (uint256 i = 0; i < d.hotWallets.length; i++) {
            address w = d.hotWallets[i];
            require(w != address(0) && w.code.length > 0, "bad hot");
            require(IGuardState(oldGuard).hotWallets(w), "hot not in old");
            require(!coldWallets[w], "hot also cold");
            for (uint256 j = 0; j < i; j++) require(d.hotWallets[j] != w, "dup hot");
            hotWallets[w] = true;
            emit HotWalletUpdated(w, true);
        }

        // ---- Cold wallets ----
        for (uint256 i = 0; i < d.coldWallets.length; i++) {
            address w = d.coldWallets[i];
            require(w != address(0) && w.code.length > 0, "bad cold");
            require(IGuardState(oldGuard).coldWallets(w), "cold not in old");
            require(!hotWallets[w], "cold also hot");
            for (uint256 j = 0; j < i; j++) require(d.coldWallets[j] != w, "dup cold");
            coldWallets[w] = true;
            emit ColdWalletUpdated(w, true);
        }

        // ---- Tokens ----
        for (uint256 i = 0; i < d.approvedTokens.length; i++) {
            address t = d.approvedTokens[i];
            require(t.code.length > 0, "bad token");
            require(IGuardState(oldGuard).approvedToken(t), "token not in old");
            for (uint256 j = 0; j < i; j++) require(d.approvedTokens[j] != t, "dup token");
            approvedToken[t] = true;
            emit TokenUpdated(t, true);
        }

        // ---- Routers ----
        for (uint256 i = 0; i < d.approvedRouters.length; i++) {
            address r = d.approvedRouters[i];
            require(r.code.length > 0, "bad router");
            require(IGuardState(oldGuard).approvedRouter(r), "router not in old");
            for (uint256 j = 0; j < i; j++) require(d.approvedRouters[j] != r, "dup router");
            approvedRouter[r] = true;
            emit RouterUpdated(r, true);
        }

        // ---- Blacklist ----
        for (uint256 i = 0; i < d.blacklistAddrs.length; i++) {
            address b = d.blacklistAddrs[i];
            require(IGuardState(oldGuard).blacklist(b), "blacklist not in old");
            for (uint256 j = 0; j < i; j++) require(d.blacklistAddrs[j] != b, "dup blacklist");
            blacklist[b] = true;
            emit BlacklistUpdated(b, true);
        }

        // ---- Whitelist ----
        for (uint256 i = 0; i < d.whitelistEntries.length; i++) {
            WhitelistMigrationEntry memory e = d.whitelistEntries[i];
            (bool ex, uint256 lm, uint256 rm, uint256 dl) =
                IGuardState(oldGuard).getWhitelistEntry(e.recipient, e.token);
            require(ex, "wl not in old");
            require(lm == e.limit && rm == e.remaining && dl == e.deadline, "wl mismatch");
            if (e.token != NATIVE) require(approvedToken[e.token], "wl token not approved");
            whitelist[e.recipient][e.token] = WhitelistEntry(true, e.limit, e.remaining, e.deadline);
            emit WhitelistUpdated(e.recipient, e.token, e.limit, e.deadline, true);
        }

        // ---- Recovery targets ----
        for (uint256 i = 0; i < d.recoveryEntries.length; i++) {
            RecoveryTargetMigrationEntry memory e = d.recoveryEntries[i];
            (bool ex, address rc) = IGuardState(oldGuard).getRecoveryTarget(e.wallet, e.token);
            require(ex, "rc not in old");
            require(rc == e.recipient, "rc mismatch");
            require(_isRegisteredWallet(e.wallet), "rc wallet not registered");
            recoveryTargets[e.wallet][e.token] = RecoveryTarget(true, e.recipient);
            emit RecoveryTargetUpdated(e.wallet, e.token, e.recipient, true);
        }

        // ---- Migration emergency key ----
        if (d.migrationEmergencyKey != address(0)) {
            require(
                d.migrationEmergencyKey == IGuardState(oldGuard).migrationEmergencyKey(),
                "emergency key mismatch"
            );
            require(d.migrationEmergencyKey.code.length == 0, "emergency not EOA");
            migrationEmergencyKey = d.migrationEmergencyKey;
            emit MigrationEmergencyKeyChanged(address(0), d.migrationEmergencyKey);
        }

        // ---- Subscription (v5.1.8: БЕЗУСЛОВНЫЙ cross-check) ----
        {
            (uint8 oldPlan, uint64 oldExpiry) = IGuardState(oldGuard).getRawSubscription();
            require(uint8(d.subscriptionPlan) == oldPlan, "sub plan mismatch");
            require(d.subscriptionExpiresAt == oldExpiry, "sub expiry mismatch");
            if (d.subscriptionExpiresAt > 0) {
                subscriptions[mainWallet] = Subscription(d.subscriptionPlan, d.subscriptionExpiresAt);
                emit Subscribed(mainWallet, d.subscriptionPlan, address(0), 0, d.subscriptionExpiresAt);
            }
        }

        // ---- Treasury (v5.1.8: БЕЗУСЛОВНЫЙ cross-check) ----
        {
            address oldTreasury = IGuardState(oldGuard).treasury();
            require(d.treasury == oldTreasury, "treasury mismatch");
            if (d.treasury != address(0)) {
                treasury = d.treasury;
                emit TreasuryUpdated(address(0), d.treasury);
            }
        }

        // ---- App version (v5.1.8: БЕЗУСЛОВНЫЙ cross-check) ----
        {
            bytes32 oldHash = IGuardState(oldGuard).appVersionHash();
            require(d.appVersionHash == oldHash, "app version mismatch");
            if (d.appVersionHash != bytes32(0)) {
                appVersionHash = d.appVersionHash;
                emit AppVersionUpdated(bytes32(0), d.appVersionHash);
            }
        }

        // ---- P2P config (v5.1.8: БЕЗУСЛОВНЫЙ cross-check) ----
        {
            address oldAdmin = IGuardState(oldGuard).p2pAdmin();
            address oldDAO = IGuardState(oldGuard).p2pDAO();
            uint64 oldTimeout = IGuardState(oldGuard).p2pTimeout();
            require(d.p2pAdmin == oldAdmin, "p2pAdmin mismatch");
            require(d.p2pDAO == oldDAO, "p2pDAO mismatch");
            require(d.p2pTimeout == oldTimeout, "p2pTimeout mismatch");
            if (d.p2pAdmin != address(0) || d.p2pDAO != address(0) || d.p2pTimeout != 0) {
                p2pAdmin = d.p2pAdmin;
                p2pDAO = d.p2pDAO;
                p2pTimeout = d.p2pTimeout;
                emit P2PConfigUpdated(d.p2pAdmin, d.p2pDAO, d.p2pTimeout);
            }
        }

        // ---- Trading platforms ----
        for (uint256 i = 0; i < d.approvedTradingPlatforms.length; i++) {
            address p = d.approvedTradingPlatforms[i];
            require(p != address(0) && p.code.length > 0, "bad platform");
            require(IGuardState(oldGuard).approvedTradingPlatform(p), "platform not in old");
            for (uint256 j = 0; j < i; j++) require(d.approvedTradingPlatforms[j] != p, "dup platform");
            approvedTradingPlatform[p] = true;
            emit TradingPlatformUpdated(p, true);
        }

        // ---- Plan configs ----
        for (uint256 i = 0; i < d.planConfigEntries.length; i++) {
            PlanConfigMigrationEntry memory e = d.planConfigEntries[i];
            require(e.plan == uint8(SubscriptionPlan.STANDARD) || e.plan == uint8(SubscriptionPlan.PREMIUM), "bad plan");
            (address pt, uint256 pr, uint64 pd, bool en) = IGuardState(oldGuard).getPlanConfig(e.plan);
            require(
                pt == e.paymentToken && pr == e.price && pd == e.period && en == e.enabled,
                "plan config mismatch"
            );
            SubscriptionPlan sp = SubscriptionPlan(e.plan);
            planConfigs[sp] = PlanConfig(e.paymentToken, e.price, e.period, e.enabled);
            emit PlanConfigUpdated(sp, e.paymentToken, e.price, e.period, e.enabled);
        }

        migrationAccepted = true;
        emit MigrationAccepted(oldGuard);
    }
