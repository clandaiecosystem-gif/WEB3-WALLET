pragma solidity ^0.8.20;

interface IServicePermit2 {
    function transferFrom(address from, address to, uint160 amount, address token) external;
    function allowance(address owner, address token, address spender)
        external view returns (uint160 amount, uint48 expiration, uint48 nonce);
}

interface IServiceWallet {
    function guard() external view returns (address);
    function permit2() external view returns (address);
    function paused() external view returns (bool);
}

interface IServiceCore {
    function mainWallet() external view returns (address);
    function permit2() external view returns (address);
    function migratedOut() external view returns (bool);
    function migrationEmergencyKey() external view returns (address);
    function isRegisteredWallet(address wallet) external view returns (bool);
    function approvedToken(address token) external view returns (bool);
    function blacklist(address target) external view returns (bool);
    function verifyOwnerSignatures(bytes32 digest, bytes[] calldata sigs, uint256 threshold, bool requireCold)
        external view returns (bool);
}

contract SmartWalletService {
    string public constant VERSION = "1.0.0";
    uint256 public constant HOT_THRESHOLD = 2;
    uint256 public constant CRITICAL_THRESHOLD = 3;
    uint256 public constant MIGRATION_DELAY = 48 hours;

    uint256 public constant PERM_APP_VERIFY = 1 << 0;
    uint256 public constant PERM_EMERGENCY = 1 << 1;
    uint256 public constant PERM_COLD_WARM = 1 << 2;
    uint256 public constant PERM_P2P = 1 << 3;
    uint256 public constant PERM_SECURITY = 1 << 4;
    uint256 public constant PERM_TRADING = 1 << 5;

    bytes32 private constant DOMAIN_TYPEHASH =
        keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)");
    bytes32 private constant SUBSCRIBE_TYPEHASH = keccak256(
        "Subscribe(address mainWallet,uint8 plan,address paymentToken,uint256 price,address fromWallet,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SET_TREASURY_TYPEHASH = keccak256(
        "SetTreasury(address mainWallet,address treasury,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SET_PLAN_CONFIG_TYPEHASH = keccak256(
        "SetPlanConfig(address mainWallet,uint8 plan,address paymentToken,uint256 price,uint64 period,bool enabled,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SET_APP_VERSION_TYPEHASH = keccak256(
        "SetAppVersion(address mainWallet,bytes32 versionHash,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant TRADING_PLATFORM_TYPEHASH = keccak256(
        "TradingPlatformOp(uint8 op,address platform,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SET_P2P_CONFIG_TYPEHASH = keccak256(
        "SetP2PConfig(address mainWallet,address p2pAdmin,address p2pDAO,uint64 timeout,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SERVICE_MIGRATION_TYPEHASH = keccak256(
        "ServiceMigration(address mainWallet,address newService,bytes32 migrationDataHash,uint256 nonce,uint256 executeAfter,uint256 deadline)"
    );
    bytes32 private constant SERVICE_MIGRATION_EMERGENCY_TYPEHASH = keccak256(
        "ServiceMigrationEmergency(address mainWallet,address newService,bytes32 migrationDataHash,uint256 nonce,uint256 executeAfter,uint256 deadline)"
    );
    bytes32 private constant CANCEL_SERVICE_MIGRATION_TYPEHASH = keccak256(
        "CancelServiceMigration(address mainWallet,address newService,uint256 nonce,uint256 deadline)"
    );
    bytes32 private constant SECP256K1N_HALF =
        0x7fffffffffffffffffffffffffffffff5d576e7357a4501ddfe92f46681b20a0;

    enum SubscriptionPlan { FREE, STANDARD, PREMIUM }
    struct Subscription { SubscriptionPlan plan; uint64 expiresAt; }
    struct PlanConfig {
        address paymentToken;
        uint256 price;
        uint64 period;
        bool enabled;
    }
    struct ServiceMigration {
        address newService;
        bytes32 migrationDataHash;
        uint64 executeAfter;
        bool pending;
    }
    struct ServiceMigrationData {
        Subscription subscription;
        address treasury;
        PlanConfig standardConfig;
        PlanConfig premiumConfig;
        bytes32 appVersionHash;
        address[] approvedTradingPlatforms;
        address p2pAdmin;
        address p2pDAO;
        uint64 p2pTimeout;
    }

    IServiceCore public immutable core;
    address public immutable mainWallet;
    address public immutable permit2;
    address public immutable expectedOldService;
    bytes32 public immutable DOMAIN_SEPARATOR;

    mapping(address => Subscription) public subscriptions;
    mapping(SubscriptionPlan => PlanConfig) public planConfigs;
    mapping(address => mapping(uint256 => bool)) public usedNonces;
    mapping(address => bool) public approvedTradingPlatform;
    address[] private _approvedPlatforms;
    mapping(address => uint256) private _platformIndexPlusOne;

    address public treasury;
    bytes32 public appVersionHash;
    address public p2pAdmin;
    address public p2pDAO;
    uint64 public p2pTimeout;

    bool public migratedOut;
    bool public migrationAccepted;
    address public migratedTo;
    ServiceMigration public pendingMigration;

    uint256 private _locked = 1;
    modifier nonReentrant() {
        require(_locked == 1, "reentrant");
        _locked = 2;
        _;
        _locked = 1;
    }
    modifier notMigrated() {
        require(!migratedOut, "migrated out");
        require(expectedOldService == address(0) || migrationAccepted, "service migration not accepted");
        _;
    }

    event Subscribed(address indexed account, SubscriptionPlan plan, address paymentToken, uint256 price, uint64 expiresAt, address indexed fromWallet);
    event TreasuryUpdated(address indexed oldTreasury, address indexed newTreasury);
    event PlanConfigUpdated(SubscriptionPlan plan, address paymentToken, uint256 price, uint64 period, bool enabled);
    event AppVersionUpdated(bytes32 indexed oldHash, bytes32 indexed newHash);
    event TradingPlatformUpdated(address indexed platform, bool active);
    event P2PConfigUpdated(address indexed admin, address indexed dao, uint64 timeout);
    event ServiceMigrationProposed(address indexed oldService, address indexed newService, bytes32 migrationDataHash, uint64 executeAfter);
    event ServiceMigrationCancelled(address indexed oldService, address indexed newService);
    event ServiceMigrationFinalized(address indexed oldService, address indexed newService);
    event ServiceMigrationAccepted(address indexed oldService);

    constructor(address core_, address expectedOldService_) {
        require(core_ != address(0) && core_.code.length > 0, "bad core");
        core = IServiceCore(core_);
        mainWallet = IServiceCore(core_).mainWallet();
        permit2 = IServiceCore(core_).permit2();
        require(mainWallet != address(0) && permit2 != address(0), "bad core config");
        expectedOldService = expectedOldService_;
        DOMAIN_SEPARATOR = keccak256(abi.encode(
            DOMAIN_TYPEHASH,
            keccak256("SmartWalletService"),
            keccak256(bytes(VERSION)),
            block.chainid,
            address(this)
        ));
    }

    function _hashTypedData(bytes32 structHash) internal view returns (bytes32) {
        return keccak256(abi.encodePacked("\x19\x01", DOMAIN_SEPARATOR, structHash));
    }
    function _consumeNonce(address scope, uint256 nonce) internal {
        require(!usedNonces[scope][nonce], "nonce used");
        usedNonces[scope][nonce] = true;
    }
    function _checkDeadline(uint256 deadline) internal view {
        require(block.timestamp <= deadline, "expired");
    }
    function _verify(bytes32 digest, bytes[] calldata signatures, bool critical) internal view {
        require(core.verifyOwnerSignatures(
            digest,
            signatures,
            critical ? CRITICAL_THRESHOLD : HOT_THRESHOLD,
            critical
        ), critical ? "invalid critical sigs" : "invalid hot sigs");
    }
    function _activePlan(address account) internal view returns (SubscriptionPlan) {
        if (account != mainWallet) return SubscriptionPlan.FREE;
        Subscription memory sub = subscriptions[account];
        if (sub.expiresAt == 0 || block.timestamp >= sub.expiresAt) return SubscriptionPlan.FREE;
        return sub.plan;
    }
    function _planMask(SubscriptionPlan plan) internal pure returns (uint256) {
        if (plan == SubscriptionPlan.FREE) return PERM_APP_VERIFY | PERM_EMERGENCY;
        return PERM_APP_VERIFY | PERM_EMERGENCY | PERM_COLD_WARM | PERM_P2P | PERM_SECURITY | PERM_TRADING;
    }
    function _checkPaymentWallet(address wallet) internal view {
        require(core.isRegisteredWallet(wallet), "wallet not registered");
        require(wallet.code.length > 0, "wallet not contract");
        require(IServiceWallet(wallet).guard() == address(core), "wrong wallet guard");
        require(IServiceWallet(wallet).permit2() == permit2, "wrong wallet permit2");
        require(!IServiceWallet(wallet).paused(), "wallet paused");
    }
    function _checkPermit2(address wallet, address token, uint256 amount) internal view {
        (uint160 allowed, uint48 expiration, ) = IServicePermit2(permit2).allowance(wallet, token, address(this));
        require(amount <= allowed, "insufficient permit2 allowance");
        require(expiration != 0 && uint256(expiration) >= block.timestamp, "permit2 allowance expired");
    }
    function _planConfigMatches(address oldService, SubscriptionPlan plan, PlanConfig memory candidate)
        internal view returns (bool)
    {
        (address token, uint256 price, uint64 period, bool enabled) =
            IServiceMigrationSource(oldService).planConfigs(plan);
        return candidate.paymentToken == token && candidate.price == price
            && candidate.period == period && candidate.enabled == enabled;
    }

    function activePlan(address account) public view returns (SubscriptionPlan) {
        return _activePlan(account);
    }
    function hasPermission(address account, uint256 permission) external view returns (bool) {
        return permission != 0 && (_planMask(_activePlan(account)) & permission) == permission;
    }
    function getSubscription(address account) external view returns (SubscriptionPlan plan, uint64 expiresAt) {
        Subscription memory sub = subscriptions[account];
        return (_activePlan(account), sub.expiresAt);
    }
    function getPlanConfig(SubscriptionPlan plan)
        external view returns (address paymentToken, uint256 price, uint64 period, bool enabled)
    {
        PlanConfig memory config = planConfigs[plan];
        return (config.paymentToken, config.price, config.period, config.enabled);
    }
    function getApprovedTradingPlatforms() external view returns (address[] memory) {
        return _approvedPlatforms;
    }

    function subscribe(
        SubscriptionPlan plan, address fromWallet, uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external nonReentrant notMigrated {
        require(plan == SubscriptionPlan.STANDARD || plan == SubscriptionPlan.PREMIUM, "bad plan");
        require(treasury != address(0), "treasury not set");
        PlanConfig memory config = planConfigs[plan];
        require(config.enabled && config.paymentToken != address(0), "plan disabled");
        require(config.price > 0 && config.price <= type(uint160).max && config.period > 0, "bad plan config");
        require(core.approvedToken(config.paymentToken), "payment token not approved");
        require(!core.blacklist(config.paymentToken), "payment token blacklisted");
        _checkPaymentWallet(fromWallet);
        _checkDeadline(deadline);
        _consumeNonce(fromWallet, nonce);
        _checkPermit2(fromWallet, config.paymentToken, config.price);

        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SUBSCRIBE_TYPEHASH, mainWallet, uint8(plan), config.paymentToken, config.price,
            fromWallet, nonce, deadline
        )));
        _verify(digest, signatures, false);

        Subscription memory oldSub = subscriptions[mainWallet];
        uint256 nextExpiry;
        if (oldSub.expiresAt > block.timestamp && oldSub.plan == plan) {
            nextExpiry = uint256(oldSub.expiresAt) + config.period;
        } else {
            nextExpiry = block.timestamp + config.period;
        }
        require(nextExpiry <= type(uint64).max, "subscription expiry overflow");
        subscriptions[mainWallet] = Subscription(plan, uint64(nextExpiry));
        IServicePermit2(permit2).transferFrom(fromWallet, treasury, uint160(config.price), config.paymentToken);
        emit Subscribed(mainWallet, plan, config.paymentToken, config.price, uint64(nextExpiry), fromWallet);
    }

    function setTreasury(address newTreasury, uint256 nonce, uint256 deadline, bytes[] calldata signatures)
        external notMigrated
    {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newTreasury != address(0), "zero treasury");
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SET_TREASURY_TYPEHASH, mainWallet, newTreasury, nonce, deadline
        )));
        _verify(digest, signatures, true);
        address oldTreasury = treasury;
        treasury = newTreasury;
        emit TreasuryUpdated(oldTreasury, newTreasury);
    }

    function setPlanConfig(
        SubscriptionPlan plan, address paymentToken, uint256 price, uint64 period, bool enabled,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(plan != SubscriptionPlan.FREE, "FREE not configurable");
        if (enabled) {
            require(paymentToken != address(0) && paymentToken.code.length > 0, "bad payment token");
            require(core.approvedToken(paymentToken) && !core.blacklist(paymentToken), "payment token unavailable");
            require(price > 0 && price <= type(uint160).max && period > 0, "bad plan config");
        }
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SET_PLAN_CONFIG_TYPEHASH, mainWallet, uint8(plan), paymentToken, price, period, enabled, nonce, deadline
        )));
        _verify(digest, signatures, true);
        planConfigs[plan] = PlanConfig(paymentToken, price, period, enabled);
        emit PlanConfigUpdated(plan, paymentToken, price, period, enabled);
    }

    function setAppVersionHash(bytes32 newHash, uint256 nonce, uint256 deadline, bytes[] calldata signatures)
        external notMigrated
    {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SET_APP_VERSION_TYPEHASH, mainWallet, newHash, nonce, deadline
        )));
        _verify(digest, signatures, false);
        bytes32 oldHash = appVersionHash;
        appVersionHash = newHash;
        emit AppVersionUpdated(oldHash, newHash);
    }

    function addTradingPlatform(address platform, uint256 nonce, uint256 deadline, bytes[] calldata signatures)
        external notMigrated
    {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(platform != address(0) && platform.code.length > 0, "bad platform");
        require(!approvedTradingPlatform[platform] && !core.blacklist(platform), "platform unavailable");
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            TRADING_PLATFORM_TYPEHASH, uint8(30), platform, nonce, deadline
        )));
        _verify(digest, signatures, false);
        approvedTradingPlatform[platform] = true;
        _approvedPlatforms.push(platform);
        _platformIndexPlusOne[platform] = _approvedPlatforms.length;
        emit TradingPlatformUpdated(platform, true);
    }

    function removeTradingPlatform(address platform, uint256 nonce, uint256 deadline, bytes[] calldata signatures)
        external notMigrated
    {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(approvedTradingPlatform[platform], "platform not approved");
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            TRADING_PLATFORM_TYPEHASH, uint8(31), platform, nonce, deadline
        )));
        _verify(digest, signatures, false);
        approvedTradingPlatform[platform] = false;
        uint256 index = _platformIndexPlusOne[platform] - 1;
        uint256 last = _approvedPlatforms.length - 1;
        if (index != last) {
            address moved = _approvedPlatforms[last];
            _approvedPlatforms[index] = moved;
            _platformIndexPlusOne[moved] = index + 1;
        }
        _approvedPlatforms.pop();
        delete _platformIndexPlusOne[platform];
        emit TradingPlatformUpdated(platform, false);
    }

    function setP2PConfig(
        address newAdmin, address newDAO, uint64 newTimeout,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newAdmin != address(0) && newDAO != address(0) && newTimeout > 0, "bad p2p config");
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SET_P2P_CONFIG_TYPEHASH, mainWallet, newAdmin, newDAO, newTimeout, nonce, deadline
        )));
        _verify(digest, signatures, true);
        p2pAdmin = newAdmin;
        p2pDAO = newDAO;
        p2pTimeout = newTimeout;
        emit P2PConfigUpdated(newAdmin, newDAO, newTimeout);
    }

    function proposeMigration(
        address newService, bytes32 migrationDataHash,
        uint256 nonce, uint256 deadline, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newService != address(0) && newService != address(this) && newService.code.length > 0, "bad service");
        require(!pendingMigration.pending, "migration pending");
        require(block.timestamp + MIGRATION_DELAY <= type(uint64).max, "migration time overflow");
        uint64 executeAfter = uint64(block.timestamp + MIGRATION_DELAY);
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SERVICE_MIGRATION_TYPEHASH, mainWallet, newService, migrationDataHash, nonce, executeAfter, deadline
        )));
        _verify(digest, signatures, true);
        pendingMigration = ServiceMigration(newService, migrationDataHash, executeAfter, true);
        emit ServiceMigrationProposed(address(this), newService, migrationDataHash, executeAfter);
    }

    function cancelMigration(uint256 nonce, uint256 deadline, bytes[] calldata signatures) external notMigrated {
        require(pendingMigration.pending && block.timestamp < pendingMigration.executeAfter, "no cancellable migration");
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            CANCEL_SERVICE_MIGRATION_TYPEHASH, mainWallet, pendingMigration.newService, nonce, deadline
        )));
        _verify(digest, signatures, true);
        address newService = pendingMigration.newService;
        delete pendingMigration;
        emit ServiceMigrationCancelled(address(this), newService);
    }

    function proposeMigrationWithEmergency(
        address newService, bytes32 migrationDataHash, uint256 nonce, uint256 deadline,
        bytes calldata emergencySignature, bytes[] calldata signatures
    ) external notMigrated {
        _checkDeadline(deadline);
        _consumeNonce(mainWallet, nonce);
        require(newService != address(0) && newService != address(this) && newService.code.length > 0, "bad service");
        require(!pendingMigration.pending, "migration pending");
        address key = core.migrationEmergencyKey();
        require(key != address(0), "emergency key not set");
        require(_recoverEmergencyKey(emergencySignature, _hashTypedData(keccak256(abi.encode(
            SERVICE_MIGRATION_EMERGENCY_TYPEHASH, mainWallet, newService, migrationDataHash, nonce, block.timestamp, deadline
        )))) == key, "invalid emergency key");
        uint64 executeAfter = uint64(block.timestamp);
        bytes32 digest = _hashTypedData(keccak256(abi.encode(
            SERVICE_MIGRATION_EMERGENCY_TYPEHASH, mainWallet, newService, migrationDataHash, nonce, executeAfter, deadline
        )));
        _verify(digest, signatures, true);
        pendingMigration = ServiceMigration(newService, migrationDataHash, executeAfter, true);
        emit ServiceMigrationProposed(address(this), newService, migrationDataHash, executeAfter);
    }

    function _recoverEmergencyKey(bytes calldata sig, bytes32 digest) internal pure returns (address signer) {
        require(sig.length == 65, "bad sig length");
        bytes32 r; bytes32 s; uint8 v;
        assembly ("memory-safe") {
            r := calldataload(sig.offset)
            s := calldataload(add(sig.offset, 32))
            v := byte(0, calldataload(add(sig.offset, 64)))
        }
        if (v < 27) v += 27;
        require((v == 27 || v == 28) && uint256(s) <= uint256(SECP256K1N_HALF), "bad emergency signature");
        signer = ecrecover(digest, v, r, s);
        require(signer != address(0), "invalid emergency signer");
    }

    function finalizeMigration(bytes calldata migrationData) external nonReentrant notMigrated {
        require(pendingMigration.pending && block.timestamp >= pendingMigration.executeAfter, "migration timelock active");
        require(keccak256(migrationData) == pendingMigration.migrationDataHash, "migration hash mismatch");
        address newService = pendingMigration.newService;
        migratedOut = true;
        migratedTo = newService;
        delete pendingMigration;
        _lastMigrationHash = keccak256(migrationData);
        IServiceMigrationTarget(newService).acceptServiceMigration(address(this), _lastMigrationHash, migrationData);
        emit ServiceMigrationFinalized(address(this), newService);
    }

    function acceptServiceMigration(address oldService, bytes32 migrationDataHash, bytes calldata migrationData) external nonReentrant {
        require(!migrationAccepted && !migratedOut, "service unavailable");
        require(expectedOldService != address(0) && oldService == expectedOldService && msg.sender == expectedOldService, "unexpected old service");
        require(keccak256(migrationData) == migrationDataHash, "migration hash mismatch");
        require(IServiceMigrationSource(oldService).migratedOut(), "old service not migrated");
        require(IServiceMigrationSource(oldService).migratedTo() == address(this), "wrong service destination");
        // The old service marks itself migrated before this atomic call; the snapshot is verified against its state.
        ServiceMigrationData memory data = abi.decode(migrationData, (ServiceMigrationData));
        require(IServiceCore(oldService).mainWallet() == mainWallet, "main wallet mismatch");
        require(IServiceCore(oldService).permit2() == permit2, "permit2 mismatch");
        require(data.subscription.plan == IServiceMigrationSource(oldService).rawSubscriptionPlan(), "subscription plan mismatch");
        require(data.subscription.expiresAt == IServiceMigrationSource(oldService).rawSubscriptionExpiry(), "subscription expiry mismatch");
        require(data.treasury == IServiceMigrationSource(oldService).treasury(), "treasury mismatch");
        require(_planConfigMatches(oldService, SubscriptionPlan.STANDARD, data.standardConfig), "standard config mismatch");
        require(_planConfigMatches(oldService, SubscriptionPlan.PREMIUM, data.premiumConfig), "premium config mismatch");
        require(data.appVersionHash == IServiceMigrationSource(oldService).appVersionHash(), "app hash mismatch");
        require(data.p2pAdmin == IServiceMigrationSource(oldService).p2pAdmin(), "p2p admin mismatch");
        require(data.p2pDAO == IServiceMigrationSource(oldService).p2pDAO(), "p2p dao mismatch");
        require(data.p2pTimeout == IServiceMigrationSource(oldService).p2pTimeout(), "p2p timeout mismatch");
        address[] memory oldPlatforms = IServiceMigrationSource(oldService).getApprovedTradingPlatforms();
        require(data.approvedTradingPlatforms.length == oldPlatforms.length, "platform count mismatch");
        for (uint256 i; i < oldPlatforms.length; ++i) {
            bool found;
            for (uint256 j; j < data.approvedTradingPlatforms.length; ++j) {
                if (oldPlatforms[i] == data.approvedTradingPlatforms[j]) { found = true; break; }
            }
            require(found, "platform missing");
        }
        subscriptions[mainWallet] = data.subscription;
        treasury = data.treasury;
        planConfigs[SubscriptionPlan.STANDARD] = data.standardConfig;
        planConfigs[SubscriptionPlan.PREMIUM] = data.premiumConfig;
        appVersionHash = data.appVersionHash;
        p2pAdmin = data.p2pAdmin;
        p2pDAO = data.p2pDAO;
        p2pTimeout = data.p2pTimeout;
        for (uint256 i; i < data.approvedTradingPlatforms.length; ++i) {
            address platform = data.approvedTradingPlatforms[i];
            require(platform.code.length > 0 && !core.blacklist(platform), "invalid platform");
            require(!approvedTradingPlatform[platform], "duplicate platform");
            approvedTradingPlatform[platform] = true;
            _approvedPlatforms.push(platform);
            _platformIndexPlusOne[platform] = _approvedPlatforms.length;
            emit TradingPlatformUpdated(platform, true);
        }
        migrationAccepted = true;
        emit ServiceMigrationAccepted(oldService);
    }

    bytes32 private _lastMigrationHash;

    function rawSubscriptionPlan() external view returns (SubscriptionPlan) { return subscriptions[mainWallet].plan; }
    function rawSubscriptionExpiry() external view returns (uint64) { return subscriptions[mainWallet].expiresAt; }

    function _migrationSnapshot() internal view returns (bytes memory) {
        return abi.encode(ServiceMigrationData(
            subscriptions[mainWallet], treasury,
            planConfigs[SubscriptionPlan.STANDARD], planConfigs[SubscriptionPlan.PREMIUM],
            appVersionHash, _approvedPlatforms, p2pAdmin, p2pDAO, p2pTimeout
        ));
    }

    function getMigrationData() external view returns (bytes memory) {
        return _migrationSnapshot();
    }
}

interface IServiceMigrationTarget {
    function acceptServiceMigration(address oldService, bytes32 migrationDataHash, bytes calldata migrationData) external;
}
interface IServiceMigrationSource {
    function migratedOut() external view returns (bool);
    function migratedTo() external view returns (address);
    function mainWallet() external view returns (address);
    function treasury() external view returns (address);
    function planConfigs(SmartWalletService.SubscriptionPlan) external view returns (address,uint256,uint64,bool);
    function appVersionHash() external view returns (bytes32);
    function p2pAdmin() external view returns (address);
    function p2pDAO() external view returns (address);
    function p2pTimeout() external view returns (uint64);
    function getApprovedTradingPlatforms() external view returns (address[] memory);
    function rawSubscriptionPlan() external view returns (SmartWalletService.SubscriptionPlan);
    function rawSubscriptionExpiry() external view returns (uint64);
}
