    function test_1_WhitelistContractWorks() public {
        _grantStandard();
        _fund(address(usd), 1_000 * 1e6);
        _setContractPolicy(address(dex), false, HUNDRED, 10_000, block.timestamp + WEEK);

        uint256 before = _balanceOf(address(usd), address(dex));
        guard.executeOnContract(
            address(wallet), address(dex), address(usd), 10 * 1e6, address(wallet),
            false, 10, type(uint256).max, "", _twoHot(_onContractDigest(
                address(wallet), address(dex), address(usd), 10 * 1e6, address(wallet), false, 10, type(uint256).max
            ))
        );
        assertEq(_balanceOf(address(usd), address(dex)) - before, 10 * 1e6);
    }

    function test_2_UnknownContractBlocked() public {
        _grantStandard();
        _fund(address(usd), 1_000 * 1e6);

        guard.executeOnContract(

                address(wallet), address(stranger), address(usd), 1e6, address(wallet), false, 10, type(uint256).max
            ))
        );
    }

    function test_5_GreenTokenWithinUsdLimitWorks() public {
        _grantStandard();
        _fund(address(usd), 1_000 * 1e6);
        _setContractPolicy(address(dex), false, HUNDRED, 10_000, block.timestamp + WEEK);

        // $50 при цене $1 -> 50e6 токена (6 decimals)
        guard.executeOnContract(
            address(wallet), address(dex), address(usd), 50 * 1e6, address(wallet),
            false, 10, type(uint256).max, "", _twoHot(_onContractDigest(
                address(wallet), address(dex), address(usd), 50 * 1e6, address(wallet), false, 10, type(uint256).max
            ))
        );
        (bool exists, , uint256 limit, uint256 spent, , ) = policy.getContractPolicy(address(dex));
        assertTrue(exists);
        assertEq(limit, HUNDRED);
        assertEq(spent, 50 * DOLLAR);
        assertEq(policy.remainingUsd(address(dex)), 50 * DOLLAR);
    }

    function test_6_GreenTokenAboveUsdLimitReverts() public {

    function test_14_UnknownTokenDefaultsToStrict() public {
        _grantStandard();
        MockERC20 unknown = new MockERC20();
        _asGuardPolicyAddToken(address(unknown));
        _fund(address(unknown), 1_000 ether);
        _setContractPolicy(address(dex), false, 1_000 * DOLLAR, 10_000, block.timestamp + WEEK);

        // Неизвестный токен не получает USD-режим: категория RED по умолчанию,
        // token amount policy отсутствует -> операция отклоняется.
        assertEq(uint256(policy.effectiveTokenRisk(address(unknown))), uint256(SmartWalletPolicyRegistry.TokenRiskCategory.RED));

        vm.expectRevert(SmartWalletPolicyRegistry.LimitExceeded.selector);
        guard.executeOnContract(
            address(wallet), address(dex), address(unknown), 1 ether, address(wallet),
            false, 10, type(uint256).max, "", _twoHot(_onContractDigest(
                address(wallet), address(dex), address(unknown), 1 ether, address(wallet), false, 10, type(uint256).max
            ))
        );
    }

    function test_15_CheapTokenCannotDrainWholeBalance() public {
        _grantStandard();

        MockERC20 cheap = new MockERC20();
        _asGuardPolicyAddToken(address(cheap));
        _setRisk(address(cheap), SmartWalletPolicyRegistry.TokenRiskCategory.GREEN);
        _setPrice(address(cheap), 1e12, 18); // $0.000001 в 1e18-шкале
        _fund(address(cheap), 1_000_000_000 ether);
        // доля баланса 5% => максимум 50e6 токенов
        _setContractPolicy(address(dex), false, 1_000 * DOLLAR, 500, block.timestamp + WEEK);

        vm.expectRevert(SmartWalletPolicyRegistry.BalanceShareExceeded.selector);
        guard.executeOnContract(

            false, 10, type(uint256).max, "", _twoHot(_onContractDigest(

    function test_17_MultipleApprovalsCannotBypassPolicy() public {
        _grantStandard();
        _fund(address(usd), 1_000 * 1e6);
        _setContractPolicy(address(dex), false, HUNDRED, 10_000, block.timestamp + WEEK);

        // Первый approve на $60 проходит
        bytes32 d1 = _approveDigest(address(wallet), address(dex), address(usd), 60 * 1e6, false, 10, type(uint256).max);
        guard.approveOnContract(
            address(wallet), address(dex), address(usd), 60 * 1e6, false, 10, type(uint256).max, "", _twoHot(d1)
        );

        // Второй approve на $60 должен быть отклонён: usdSpent уже 60
        bytes32 d2 = _approveDigest(address(wallet), address(dex), address(usd), 60 * 1e6, false, 11, type(uint256).max);
        vm.expectRevert(SmartWalletPolicyRegistry.UsdLimitExceeded.selector);
        guard.approveOnContract(
            address(wallet), address(dex), address(usd), 60 * 1e6, false, 11, type(uint256).max, "", _twoHot(d2)
        );
    }
